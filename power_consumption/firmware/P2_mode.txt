/**
  ******************************************************************************
  * @file    : main.c
  * @brief   : Mode P2 - PM active, CPU in a busy loop
  *
  *            PURPOSE
  *            -------
  *            P2 is identical to M1 except that the processor core is left
  *            running instead of being placed in Sleep mode. The difference
  *            P2 - P1 quantifies the power saved by the sleep decision,
  *            with no change to the output waveform.
  *
  *            CLOCK CONFIGURATION
  *            -------------------
  *            Clock source : HSE (8 MHz, bypass, from ST-LINK MCO)
  *            PLL          : 8 MHz x 9 = 72 MHz
  *            HCLK         : 72 MHz
  *            PCLK1        : 36 MHz   (APB1 timer clocks = 72 MHz)
  *            PCLK2        : 72 MHz
  *            Flash latency: 2 wait states
  *            CPU state    : busy loop
  *
  *            TIMER CONFIGURATION (set in CubeMX, not here)
  *            --------------------------------------------
  *            TIM6 clock = 72 MHz
  *            PSC = 0, ARR = 719
  *            Fs = 72 MHz / (0 + 1)(719 + 1) = 100.000 kSa/s
  *
  *            SIGNAL PARAMETERS
  *            -----------------
  *            fc = 5 kHz, fm = 500 Hz, kp = 1.0 rad
  *
  *            Buffer length rule: N = k * lcm(Fs/fc, Fs/fm)
  *              Fs/fc = 20, Fs/fm = 200, lcm = 200
  *              N = 2000 = 10 x 200
  *              -> 100 carrier cycles, 10 message cycles
  *              -> no phase discontinuity at the DMA wrap
  *
  *            EXPECTED RESULT
  *            ---------------
  *            Oscilloscope: PM waveform, constant amplitude ~1.6 V pk-pk,
  *            cycles compressing and expanding at a 500 Hz rate.
  ******************************************************************************
  */
/* USER CODE END Header */

/* Includes ------------------------------------------------------------------*/
#include "main.h"
#include "dac.h"
#include "dma.h"
#include "tim.h"
#include "gpio.h"

/* USER CODE BEGIN Includes */
#include <math.h>
/* USER CODE END Includes */

/* Private define ------------------------------------------------------------*/
/* USER CODE BEGIN PD */

#define PI              3.14159265358979323846f

#define FS              100000.0f   /* DAC sampling frequency = 100 kSa/s */
#define FC              5000.0f     /* Carrier frequency      = 5 kHz     */
#define FM              500.0f      /* Message frequency      = 500 Hz    */
#define KP              1.0f        /* Phase sensitivity      = 1.0 rad   */

#define DAC_MID         2048        /* Midpoint for 12-bit DAC            */
#define DAC_AMP         1000        /* Amplitude, keep below clipping     */

#define DAC_BUF_SIZE    2000        /* 100 carrier cycles, 10 message cycles */

/* USER CODE END PD */

/* Private variables ---------------------------------------------------------*/
/* USER CODE BEGIN PV */

uint16_t dac_buffer[DAC_BUF_SIZE];

/* USER CODE END PV */

/* Private function prototypes -----------------------------------------------*/
void SystemClock_Config(void);

/* USER CODE BEGIN PFP */
void Generate_Analog_PM_Buffer(void);
/* USER CODE END PFP */

/* USER CODE BEGIN 0 */

/**
  * @brief  Precompute one full period of the phase-modulated waveform.
  *
  *         s(t) = Ac * cos(2*pi*fc*t + kp*m(t))
  *         m(t) = sin(2*pi*fm*t)
  *
  *         Called once, before the DAC is started. The CPU performs no
  *         waveform computation after this point.
  */
void Generate_Analog_PM_Buffer(void)
{
    for (uint32_t n = 0; n < DAC_BUF_SIZE; n++)
    {
        /* Sample time. At Fs = 100 kSa/s each sample is 10 us apart. */
        float t = (float)n / FS;

        /* Sinusoidal message signal */
        float message = sinf(2.0f * PI * FM * t);

        /* Instantaneous phase: carrier term plus phase deviation */
        float phase = (2.0f * PI * FC * t) + (KP * message);

        /* Phase-modulated output, -1 to +1 */
        float y = cosf(phase);

        /* Map to the 12-bit DAC range, centred on the midpoint */
        float dac_value = (float)DAC_MID + ((float)DAC_AMP * y);

        /* Safety clipping */
        if (dac_value < 0.0f)    { dac_value = 0.0f; }
        if (dac_value > 4095.0f) { dac_value = 4095.0f; }

        dac_buffer[n] = (uint16_t)dac_value;
    }
}

/* USER CODE END 0 */

/**
  * @brief  The application entry point.
  * @retval int
  */
int main(void)
{
    /* Reset of all peripherals, initialise the Flash interface and SysTick. */
    HAL_Init();

    /* Configure the system clock: HSE + PLL -> 72 MHz */
    SystemClock_Config();

    /* Initialise all configured peripherals */
    MX_DMA_Init();
    MX_GPIO_Init();
    MX_DAC1_Init();
    MX_TIM6_Init();

    /* USER CODE BEGIN 2 */

    /* Precompute the waveform before starting the DAC */
    Generate_Analog_PM_Buffer();

    /* Start the DAC in circular DMA mode.
     *
     * If the generated handle is hdac rather than hdac1, change it here.
     */
    if (HAL_DAC_Start_DMA(&hdac1,
                          DAC_CHANNEL_1,
                          (uint32_t *)dac_buffer,
                          DAC_BUF_SIZE,
                          DAC_ALIGN_12B_R) != HAL_OK)
    {
        Error_Handler();
    }

    /* Start TIM6. Its TRGO event triggers the DAC at 100 kSa/s. */
    if (HAL_TIM_Base_Start(&htim6) != HAL_OK)
    {
        Error_Handler();
    }

    /* USER CODE END 2 */

    /* Infinite loop */
   while (1)
    {
        /* Busy loop. The CPU clock keeps running with no useful work.
         * The output waveform is identical to M1 because TIM6, DMA and
         * the DAC operate autonomously in both modes.
         */
    }
}

/**
  * @brief  System Clock Configuration
  *
  *         HSE  = 8 MHz, supplied as a clock signal by the on-board
  *                ST-LINK MCO output. BYPASS mode is required because
  *                no crystal is fitted for HSE on the Nucleo board.
  *
  *         PLL  = HSE / PREDIV(1) x PLLMUL(9) = 72 MHz
  *
  *         Note: on the STM32F3 HAL, PREDIV is a member of the PLL
  *         sub-structure, not of RCC_OscInitTypeDef directly.
  *
  * @retval None
  */
void SystemClock_Config(void)
{
    RCC_OscInitTypeDef RCC_OscInitStruct = {0};
    RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};

    RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_HSE;
    RCC_OscInitStruct.HSEState       = RCC_HSE_BYPASS;
    RCC_OscInitStruct.PLL.PLLState   = RCC_PLL_ON;
    RCC_OscInitStruct.PLL.PLLSource  = RCC_PLLSOURCE_HSE;
    RCC_OscInitStruct.PLL.PREDIV     = RCC_PREDIV_DIV1;
    RCC_OscInitStruct.PLL.PLLMUL     = RCC_PLL_MUL9;

    if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK)
    {
        Error_Handler();
    }

    RCC_ClkInitStruct.ClockType      = RCC_CLOCKTYPE_HCLK
                                     | RCC_CLOCKTYPE_SYSCLK
                                     | RCC_CLOCKTYPE_PCLK1
                                     | RCC_CLOCKTYPE_PCLK2;
    RCC_ClkInitStruct.SYSCLKSource   = RCC_SYSCLKSOURCE_PLLCLK;
    RCC_ClkInitStruct.AHBCLKDivider  = RCC_SYSCLK_DIV1;   /* HCLK  = 72 MHz */
    RCC_ClkInitStruct.APB1CLKDivider = RCC_HCLK_DIV2;     /* PCLK1 = 36 MHz */
    RCC_ClkInitStruct.APB2CLKDivider = RCC_HCLK_DIV1;     /* PCLK2 = 72 MHz */

    /* FLASH_LATENCY_2 is mandatory at 72 MHz. */
    if (HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_2) != HAL_OK)
    {
        Error_Handler();
    }
}

/**
  * @brief  This function is executed in case of error occurrence.
  * @retval None
  */
void Error_Handler(void)
{
    __disable_irq();
    while (1)
    {
    }
}
